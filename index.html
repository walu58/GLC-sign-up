/** * Global Language Center — Meeting Sign-up * Google Apps Script backend (stores every sign-up in a Google Sheet). * * Setup: see README.md. Change ADMIN_PASSCODE before deploying. */ const ADMIN_PASSCODE = 'glcc2026'; const SLOT_CAPACITY = 1; // must match SLOT_CAPACITY in index.html const SHEET_NAME = 'signups'; function sheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['id', 'lang', 'date', 'time', 'name', 'email', 'note', 'createdAt']); sh.setFrozenRows(1); } return sh; } function rows_() { const values = sheet_().getDataRange().getValues(); return values.slice(1).filter(function (r) { return r[0]; }).map(function (r) { return { id: String(r[0]), lang: String(r[1]), date: formatDate_(r[2]), time: formatTime_(r[3]), name: String(r[4]), email: String(r[5]), note: String(r[6]) }; }); } /** Sheets may turn "2026-09-14" into a Date object; normalise it back. */ function formatDate_(v) { if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); return String(v); } function formatTime_(v) { if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), 'HH:mm'); return String(v); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** Public reads: the booking board only needs to know which slots are used. */ function doGet(e) { const action = (e.parameter.action || 'list'); if (action === 'admin') { if (e.parameter.key !== ADMIN_PASSCODE) return json_({ error: 'unauthorised' }); return json_({ bookings: rows_() }); } // Public list: names and notes are stripped so students cannot read them. const publicRows = rows_().map(function (b) { return { id: b.id, lang: b.lang, date: b.date, time: b.time, name: '', email: '', note: '' }; }); return json_({ bookings: publicRows }); } function doPost(e) { let body = {}; try { body = JSON.parse(e.postData.contents); } catch (err) { return json_({ error: 'bad request' }); } if (body.action === 'book') return book_(body); if (body.action === 'remove') return remove_(body); return json_({ error: 'unknown action' }); } function book_(b) { if (!b.lang || !b.date || !b.time || !b.name || !b.email || !b.note) { return json_({ error: 'missing fields' }); } const note = String(b.note).slice(0, 100); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const used = rows_().filter(function (r) { return r.lang === b.lang && r.date === b.date && r.time === b.time; }).length; if (used >= SLOT_CAPACITY) return json_({ taken: true }); sheet_().appendRow([ b.id || (Date.now() + '-' + Math.random().toString(36).slice(2, 7)), b.lang, "'" + b.date, "'" + b.time, b.name, b.email, note, new Date() ]); notify_(b, note); return json_({ ok: true }); } finally { lock.releaseLock(); } } function remove_(b) { if (b.key !== ADMIN_PASSCODE) return json_({ error: 'unauthorised' }); const sh = sheet_(); const values = sh.getDataRange().getValues(); for (let i = values.length - 1; i >= 1; i--) { if (String(values[i][0]) === String(b.id)) { sh.deleteRow(i + 1); return json_({ ok: true }); } } return json_({ error: 'not found' }); } /** * Optional: email the office whenever a student books. * Add an address to turn it on; leave it empty to skip. */ const NOTIFY_TO = ''; function notify_(b, note) { if (!NOTIFY_TO) return; MailApp.sendEmail({ to: NOTIFY_TO, subject: 'GLC sign-up: ' + b.lang + ' ' + b.date + ' ' + b.time, body: b.name + ' (' + b.email + ')\n' + b.lang + ' — ' + b.date + ' ' + b.time + '\n\n' + note }); }
