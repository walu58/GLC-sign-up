/** * Global Language Center — Meeting Sign-up * Google Apps Script backend. Stores every sign-up in a Google Sheet. * * All requests arrive as GET with a ?callback= parameter (JSONP), which is the * only way a GitHub Pages site can talk to Apps Script without CORS trouble. */ const ADMIN_PASSCODE = 'glcc2026'; const SLOT_CAPACITY = 1; // must match SLOT_CAPACITY in index.html const SHEET_NAME = 'signups'; const NOTIFY_TO = 'glc@taejae.ac.kr'; // every sign-up is emailed here function doGet(e) { const p = (e && e.parameter) ? e.parameter : {}; let out; try { switch (p.action) { case 'admin': out = adminList_(p); break; case 'book': out = book_(p); break; case 'remove': out = remove_(p); break; case 'ping': out = { ok: true, rows: rows_().length, capacity: SLOT_CAPACITY }; break; default: out = publicList_(); break; } } catch (err) { out = { error: String(err && err.message ? err.message : err) }; } return reply_(out, p.callback); } /** Some browsers still send a POST; treat it the same way. */ function doPost(e) { return doGet(e); } function reply_(obj, callback) { const json = JSON.stringify(obj); if (callback) { return ContentService .createTextOutput(callback + '(' + json + ');') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); } /* ------------------------------------------------------------------ sheet */ function sheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['id', 'lang', 'date', 'time', 'name', 'email', 'note', 'createdAt']); sh.setFrozenRows(1); } return sh; } function rows_() { const values = sheet_().getDataRange().getValues(); return values.slice(1).filter(function (r) { return r[0]; }).map(function (r) { return { id: String(r[0]), lang: String(r[1]), date: asDate_(r[2]), time: asTime_(r[3]), name: String(r[4]), email: String(r[5]), note: String(r[6]) }; }); } /** Sheets likes to turn "2026-09-14" into a Date; normalise it back to text. */ function asDate_(v) { if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); return String(v).trim(); } function asTime_(v) { if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), 'HH:mm'); return String(v).trim(); } /* ----------------------------------------------------------------- reads */ /** Students only need to know which times are used — no names, no notes. */ function publicList_() { const list = rows_().map(function (b) { return { id: b.id, lang: b.lang, date: b.date, time: b.time, name: '', email: '', note: '' }; }); return { bookings: list }; } function adminList_(p) { if (p.key !== ADMIN_PASSCODE) return { error: 'unauthorised' }; return { bookings: rows_() }; } /* ---------------------------------------------------------------- writes */ function book_(p) { if (!p.lang || !p.date || !p.time || !p.name || !p.email || !p.note) { return { error: 'Some details were missing. Please fill in every field.' }; } const note = String(p.note).slice(0, 100); const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const used = rows_().filter(function (r) { return r.lang === p.lang && r.date === p.date && r.time === p.time; }).length; if (used >= SLOT_CAPACITY) return { taken: true }; sheet_().appendRow([ p.id || (new Date().getTime() + '-' + Math.random().toString(36).slice(2, 7)), p.lang, "'" + p.date, // leading quote keeps Sheets from reformatting these "'" + p.time, p.name, p.email, note, new Date() ]); notify_(p, note); return { ok: true }; } finally { lock.releaseLock(); } } function remove_(p) { if (p.key !== ADMIN_PASSCODE) return { error: 'unauthorised' }; const sh = sheet_(); const values = sh.getDataRange().getValues(); for (let i = values.length - 1; i >= 1; i--) { if (String(values[i][0]) === String(p.id)) { sh.deleteRow(i + 1); return { ok: true }; } } return { error: 'That booking was not found.' }; } function notify_(p, note) { if (!NOTIFY_TO) return; try { MailApp.sendEmail({ to: NOTIFY_TO, replyTo: p.email, name: 'GLC Meeting Sign-up', subject: '[GLC] ' + p.lang + ' — ' + p.date + ' ' + p.time + ' — ' + p.name, body: 'A new meeting has been booked.\n\n' + 'Language : ' + p.lang + '\n' + 'When : ' + p.date + ' at ' + p.time + '\n' + 'Student : ' + p.name + '\n' + 'Email : ' + p.email + '\n\n' + 'What they would like to discuss:\n' + note + '\n\n' + 'Reply to this message to reach the student directly.' }); } catch (err) { // A failed notification must never block a student's booking. } }
